cmake_minimum_required(VERSION 3.8)
project(cyberdog_autonomy)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(cyberdog_msg REQUIRED)
find_package(gazebo_msgs REQUIRED)
find_package(OpenCV REQUIRED)
find_package(lcm REQUIRED)
add_executable(race_autonomy src/race_autonomy.cpp)
target_include_directories(race_autonomy PRIVATE include)
ament_target_dependencies(race_autonomy rclcpp sensor_msgs cv_bridge cyberdog_msg gazebo_msgs)
target_link_libraries(race_autonomy lcm ${OpenCV_LIBS})
install(TARGETS race_autonomy DESTINATION lib/${PROJECT_NAME})
ament_package()
